 #  Carga  y  preparación  de  datos 
df
 = pd.read_csv("pull_requests.csv")  
 #  Variables  de  entrada  (features)  y  variable  objetivo  (target) 
X
 = df[["lineas_modificadas",  "archivos_modificados",  "tiene_tests",  "modulo_con_historial_bugs"]]

 = df["riesgo_bug"]   #  columna  con  las  clases:  "alto",  "medio",  "bajo"

import pandas  as pd  from sklearn.cluster  import KMeans  from sklearn.preprocessing  import StandardScaler  import matplotlib.pyplot  as plt  
 #  Datos  de  servidores:  uso  promedio  de  CPU  y  memoria 
df
 = pd.read_csv("servidores.csv")  
X
 = df[["uso_cpu_promedio",  "uso_memoria_promedio"]]  
 #  Estandarización:  K-Means  usa  distancias  geométricas,  así  que  las #  variables  deben  estar  en  escalas  comparables 
X_escalado
 = StandardScaler().fit_transform(X)  
 #  Método  del  codo  para  elegir  K 
inercias
 = []  for k  in range(1,  8):  
    
modelo_k
 = KMeans(n_clusters=k,  random_state=42,  n_init=10)  
    
modelo_k.fit(X_escalado)
 
    
inercias.append(modelo_k.inertia_)
 
 
plt.plot(range(1,  8),  inercias,  marker="o")  
plt.xlabel("Cantidad  de  clusters  (K)")  
plt.ylabel("Inercia  (compactación  de  los  grupos)")  
plt.title("Método  del  codo")  
plt.show()
 
 #  Entrenamiento  con  el  K  elegido  (por  ejemplo,  K=3  según  el  gráfico  anterior) 
modelo
 = KMeans(n_clusters=3,  random_state=42,  n_init=10)  
df["cluster"]  = modelo.fit_predict(X_escalado)  
 #  Visualización  de  los  grupos  resultantes 
plt.scatter(df["uso_cpu_promedio"],  df["uso_memoria_promedio"],  c=df["cluster"],  
cmap="viridis")  
plt.xlabel("Uso  de  CPU  (%)")  
plt.ylabel("Uso  de  memoria  (%)")  
plt.title("Servidores  agrupados  por  perfil  de  consumo")  
plt.show()
